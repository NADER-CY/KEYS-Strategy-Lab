//@version=6
indicator("KEYS Liquidity Trap", overlay=true, max_labels_count=500)
// Inputs & Risk Parameters
length = input.int(14, "Lookback Period", minval=1)
atrMult = input.float(1.5, "ATR Volatility Buffer", step=0.1)
src = input.source(close, "Calculation Source")
filterSweeps = input.bool(true, "Filter Liquidity Sweeps")
// Liquidity & Order Block Logic
highBand = ta.highest(high, length)
lowBand = ta.lowest(low, length)
atrVal = ta.atr(14)
// Trap triggers
sweepBuy = ta.crossover(close, lowBand + (atrVal * atrMult)) and filterSweeps
sweepSell = ta.crossunder(close, highBand - (atrVal * atrMult)) and filterSweeps
// Visual signals
plotshape(sweepBuy, title="Trap Buy", style=shape.triangleup, location=location.belowbar, color=color.green, size=size.small, text="BUY")
plotshape(sweepSell, title="Trap Sell", style=shape.triangledown, location=location.abovebar, color=color.red, size=size.small, text="SELL")
